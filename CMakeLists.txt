cmake_minimum_required(VERSION 3.20)
project(LUM8Native VERSION 0.2 LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(LUM8_FETCH_DEPS "Download pinned codec dependencies" ON)
option(LUM8_WITH_ZSTD "Read upstream Zstandard streams" ON)
if(LUM8_FETCH_DEPS)
  set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(zlib GIT_REPOSITORY https://github.com/madler/zlib.git GIT_TAG v1.3.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(zlib)
  set(LUM8_ZLIB zlibstatic)
else()
  find_package(ZLIB REQUIRED)
  set(LUM8_ZLIB ZLIB::ZLIB)
endif()
add_library(lum8core STATIC src/lum8.cpp)
set_target_properties(lum8core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(lum8 src/cli.cpp)
target_link_libraries(lum8 PRIVATE lum8core)
add_library(lum8codec SHARED src/api.cpp)
target_compile_definitions(lum8codec PRIVATE LUM8_BUILD_DLL)
target_compile_features(lum8codec PRIVATE cxx_std_17)
target_link_libraries(lum8codec PRIVATE lum8core)
target_compile_features(lum8core PUBLIC cxx_std_17)
target_link_libraries(lum8core PUBLIC ${LUM8_ZLIB})
if(LUM8_WITH_ZSTD)
  if(LUM8_FETCH_DEPS)
    set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(zstd GIT_REPOSITORY https://github.com/facebook/zstd.git GIT_TAG v1.5.7 GIT_SHALLOW TRUE SOURCE_SUBDIR build/cmake)
    FetchContent_MakeAvailable(zstd)
    target_link_libraries(lum8core PUBLIC libzstd_static)
    target_include_directories(lum8core PRIVATE ${zstd_SOURCE_DIR}/lib)
  else()
    find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
    find_library(ZSTD_LIBRARY zstd REQUIRED)
    target_include_directories(lum8core PRIVATE ${ZSTD_INCLUDE_DIR})
    target_link_libraries(lum8core PUBLIC ${ZSTD_LIBRARY})
  endif()
  target_compile_definitions(lum8core PRIVATE LUM8_ZSTD)
endif()
install(TARGETS lum8 lum8codec
  RUNTIME DESTINATION bin COMPONENT LUM8
  LIBRARY DESTINATION lib COMPONENT LUM8
  ARCHIVE DESTINATION lib COMPONENT LUM8)
install(FILES include/lum8_api.h DESTINATION include COMPONENT LUM8)
